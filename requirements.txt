# Core simulation / RL stack. See README.md ("Why We Are Using These
# Libraries") for what each package is for.
#
# The original version of this file was a `pip freeze` from a CUDA-enabled
# Linux training environment (it pinned exact nvidia-cuda-*/cuda-toolkit
# wheels and torch==2.14.0). Those CUDA packages don't have matching wheels
# on Windows/CPU-only machines and aren't needed to run this project, so
# this file now lists the actual direct dependencies with sane version
# ranges instead. If you have an NVIDIA GPU and want CUDA acceleration,
# install a CUDA build of torch yourself first (see
# https://pytorch.org/get-started/locally/), then `pip install -r
# requirements.txt` for everything else.
numpy>=1.26,<3
pandas>=2.0,<3
simpy>=4.1,<5
gymnasium>=1.0,<2

# torch>=2.8 is stable-baselines3==2.9.0's minimum supported version.
# NOTE: on at least Windows + Python 3.12, SB3's model-checkpoint *loading*
# hits a real upstream bug with every torch version SB3 2.9.0 supports
# (verified with both 2.8.0 and 2.14.0) -- see the docstring in
# src/agents/_torch_load_workaround.py for the root cause and the narrow,
# automatically-applied workaround (it's imported by src/agents/__init__.py,
# no action needed to use it).
torch>=2.8,<3
stable-baselines3>=2.9,<3
tensorboard>=2.15   # optional: only used for train_qlearning.py's --log-dir TensorBoard logs

pytest>=8.0,<9
